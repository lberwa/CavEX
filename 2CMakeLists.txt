# PA_USE_ALSA=1 ist noetig, damit PortAudio das ALSA-Backend registriert
# (pa_unix_hostapis.c nimmt PaAlsa_Initialize nur bei #if PA_USE_ALSA in die
# Initializer-Liste auf). Ohne das Makro gibt es 0 Host-APIs -> 0 Geraete ->
# Pa_GetDefaultOutputDevice() == -1 -> "Device unavailable".
target_compile_definitions(cavex PRIVATE PLATFORM_PC CGLM_ALL_UNALIGNED)
if(UNIX AND NOT APPLE)
	target_compile_definitions(cavex PRIVATE PA_USE_ALSA=1)
elseif(WIN32)
	target_compile_definitions(cavex PRIVATE PA_USE_WASAPI=1 PA_USE_WMME=1)
endif()
target_include_directories(cavex PRIVATE
	source
	source/cubiomes
	source/sound/pc_sound/include
	source/sound/pc_sound/src/common
	source/sound/pc_sound/src/os/win
	source/sound/pc_sound/src/hostapi/wasapi
	source/sound/pc_sound/src/hostapi/wmme)
if(NOT MSVC)
	target_compile_options(cavex PRIVATE -fwrapv)
endif()

set_target_properties(
	cavex PROPERTIES
	C_STANDARD 99
)

set_property(TARGET cavex PROPERTY INTERPROCEDURAL_OPTIMIZATION False)

if(NOT WIN32)
	find_package(PkgConfig QUIET)
	if(PkgConfig_FOUND)
		pkg_check_modules(MPG123 QUIET libmpg123)
	endif()
	if(NOT MPG123_FOUND)
		find_path(MPG123_INCLUDE_DIRS mpg123.h)
		find_library(MPG123_LIBRARIES NAMES mpg123)
	endif()
else()
	# The cross pkg-config result points at the host/debug import paths and
	# produces an unusable -lmpg123 link. MP3 playback is optional on Windows.
	set(CAVEX_NO_MPG123 ON)
	set(MPG123_INCLUDE_DIRS "")
	set(MPG123_LIBRARIES "")
endif()
if(NOT MPG123_INCLUDE_DIRS OR NOT MPG123_LIBRARIES)
	if(UNIX AND NOT APPLE)
		message(FATAL_ERROR "libmpg123 is required on Linux (install libmpg123-dev)")
	else()
		message(STATUS "libmpg123 not found: background MP3 music disabled")
		set(CAVEX_NO_MPG123 ON)
	endif()
endif()
if(NOT CAVEX_NO_MPG123)
	target_include_directories(cavex PRIVATE ${MPG123_INCLUDE_DIRS})
endif()
target_link_libraries(cavex ${CMAKE_THREAD_LIBS_INIT} ZLIB::ZLIB ${ALSA_LIBRARIES} ${CAVEX_AUDIO_LIBS} glfw GLEW::GLEW OpenGL::GL ${MPG123_LIBRARIES})
if(CAVEX_NO_MPG123)
	target_compile_definitions(cavex PRIVATE CAVEX_NO_MPG123=1)
endif()
if(NOT WIN32 AND NOT APPLE)
	target_link_libraries(cavex m)
endif()
